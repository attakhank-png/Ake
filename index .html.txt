        // ระบบเรียก API พร้อม Exponential Backoff
        async function fetchWithBackoff(url, options, retries = 4) {
            let delay = 1000;
            for (let i = 0; i < retries; i++) {
                try {
                    const res = await fetch(url, options);
                    if (res.ok) return res;
                    if (res.status === 429 || res.status >= 500) {
                        await new Promise(r => setTimeout(r, delay));
                        delay *= 2;
                        continue;
                    }
                    const errBody = await res.json().catch(() => ({}));
                    throw new Error(errBody.error?.message || `เกิดข้อผิดพลาด (${res.status})`);
                } catch (e) {
                    if (i === retries - 1) throw e;
                    await new Promise(r => setTimeout(r, delay));
                    delay *= 2;
                }
            }
        }

        // คำนวณยอดเงินจากข้อความในกล่องแก้ไขรวม (รองรับการแก้ตัวเลขหรือราคาสดๆ)
        function recalculateFromEditor() {
            const rawText = masterEditor.value.trim();
            let sum3D = 0, count3D = 0;
            let sum2D = 0, count2D = 0;

            if (rawText) {
                // แยกเป็นชุดๆ ตามบรรทัดว่าง
                const blocks = rawText.split(/\n\s*\n/);
                blocks.forEach(block => {
                    const lines = block.trim().split('\n').map(l => l.trim()).filter(Boolean);
                    if (lines.length === 0) return;

                    let pricePerNumber = 0;
                    let numberTokens = [];

                    // ตรวจสอบบรรทัดสุดท้ายว่าเป็นราคาหรือไม่ (เช่น 50*50, 100*100*50, 10)
                    const lastLine = lines[lines.length - 1];
                    const isLastLinePrice = lines.length > 1 && (
                        lastLine.includes('*') || 
                        /[xX×]/.test(lastLine) || 
                        !/^\d{2,3}(\s+\d{2,3})*$/.test(lastLine)
                    );

                    if (isLastLinePrice) {
                        const cleanPriceStr = lastLine.replace(/[xX×]/g, '*');
                        const parts = cleanPriceStr.split('*');
                        parts.forEach(p => {
                            const num = parseFloat(p.replace(/[^\d.]/g, ''));
                            if (!isNaN(num)) pricePerNumber += num;
                        });
                        for (let i = 0; i < lines.length - 1; i++) {
                            const tokens = lines[i].split(/\s+/).filter(Boolean);
                            numberTokens.push(...tokens);
                        }
                    } else {
                        // กรณีไม่มีราคาบรรทัดล่าง ให้นับเฉพาะจำนวนตัวเลข
                        lines.forEach(line => {
                            const tokens = line.split(/\s+/).filter(Boolean);
                            numberTokens.push(...tokens);
                        });
                    }

                    numberTokens.forEach(token => {
                        const digitsOnly = token.replace(/\D/g, '');
                        if (digitsOnly.length === 3) {
                            count3D++;
                            sum3D += pricePerNumber;
                        } else if (digitsOnly.length === 2) {
                            count2D++;
                            sum2D += pricePerNumber;
                        }
                    });
                });
            }

            const total = sum3D + sum2D;
            document.getElementById('count-3d').textContent = count3D.toLocaleString();
            document.getElementById('sum-3d').textContent = sum3D.toLocaleString();
            document.getElementById('count-2d').textContent = count2D.toLocaleString();
            document.getElementById('sum-2d').textContent = sum2D.toLocaleString();
            document.getElementById('sum-total').textContent = total.toLocaleString();
        }

        // ฟังก์ชันคัดลอกข้อความลง Clipboard (รองรับทั้งมือถือและ iFrame)
        function copyTextToClipboard(text) {
            if (!text) return false;
            const textarea = document.createElement('textarea');
            textarea.value = text;
            textarea.style.position = 'fixed';
            textarea.style.left = '-9999px';
            textarea.style.top = '0';
            document.body.appendChild(textarea);
            textarea.focus();
            textarea.select();
            let ok = false;
            try {
                ok = document.execCommand('copy');
            } catch (err) {
                ok = false;
            }
            document.body.removeChild(textarea);
            return ok;
        }

            const text = masterEditor.value.trim();
            if (!text) {
                showToast('ไม่มีข้อความสำหรับส่งต่อ', true);
                return;
            }

            // คัดลอกเก็บไว้ใน Clipboard ก่อนเสมอ เผื่อผู้ใช้เลือกห้องแชทเอง
            copyTextToClipboard(text);
            showToast('กำลังเปิด LINE พร้อมข้อความ...');

            // ใช้ LINE URL Scheme เพื่อเปิดหน้าเลือกเพื่อน/กลุ่มใน LINE พร้อมข้อความอัตโนมัติ
            const encodedText = encodeURIComponent(text);
            const lineShareUrl = `https://line.me/R/share?text=${encodedText}`;
            
            setTimeout(() => {
                window.open(lineShareUrl, '_blank');
            }, 350);
        }

        // แสดงแถบแจ้งเตือนด้านบน
        let toastTimer;
        function showToast(message, isError = false) {
            const toast = document.getElementById('toast');
            const toastBox = document.getElementById('toast-box');
            const toastText = document.getElementById('toast-text');
            const toastIcon = document.getElementById('toast-icon');

            toastText.textContent = message;
            if (isError) {
                toastBox.classList.remove('bg-slate-900/95');
                toastBox.classList.add('bg-red-600/95');
                toastIcon.setAttribute('data-lucide', 'alert-circle');
                toastIcon.classList.remove('text-emerald-400');
                toastIcon.classList.add('text-white');
            } else {
                toastBox.classList.remove('bg-red-600/95');
                toastBox.classList.add('bg-slate-900/95');
                toastIcon.setAttribute('data-lucide', 'check-circle-2');
                toastIcon.classList.remove('text-white');
                toastIcon.classList.add('text-emerald-400');
            }
            lucide.createIcons();

            toast.classList.remove('-translate-y-6', 'opacity-0');
            clearTimeout(toastTimer);
            toastTimer = setTimeout(() => {
                toast.classList.add('-translate-y-6', 'opacity-0');
            }, 2600);
        }

        function escapeHtml(str) {
            return str
                .replace(/&/g, "&amp;")
                .replace(/</g, "&lt;")
                .replace(/>/g, "&gt;")
                .replace(/"/g, "&quot;")
                .replace(/'/g, "&#039;");
        }
